飞牛 OS Docker 与多人在线编辑

自动安装会直接创建两个 Docker 容器（OnlyOffice 与回调中继），均使用 restart=always，不会创建 Compose 项目。默认优先使用国内镜像代理；OnlyOffice 端口映射到所有网卡，可按需通过飞牛防火墙控制访问。

飞牛 OS 手动部署教程

  1. 打开飞牛桌面“Docker”，进入“Compose”，点击“新增项目”。
  2. 项目名称填写 fnoffice-onlyoffice，将下方 YAML 粘贴到项目配置中。
  3. 把 ONLYOFFICE_PORT 改成设置页显示的内部端口，并设置随机的 JWT_SECRET。
  4. 确认项目目录位于飞牛可读写的存储空间，点击创建并启动；端口会监听所有网卡，请按需配置飞牛防火墙或路由访问策略。示例已启用私有地址访问，确保 OnlyOffice 能读取 FnOffice 的 Docker 内部文件中继。
  5. 回到本页填写相同的 OnlyOffice 地址、端口和 JWT Secret，保存后再打开文档。
services:
  onlyoffice:
    image: ${ONLYOFFICE_IMAGE:-docker.m.daocloud.io/onlyoffice/documentserver:latest}
    restart: always
    ports:
      - "${ONLYOFFICE_PORT}:80"
    environment:
      JWT_ENABLED: "true"
      JWT_SECRET: "${JWT_SECRET}"
      JWT_HEADER: "Authorization"
      JWT_IN_BODY: "false"
      ALLOW_PRIVATE_IP_ADDRESS: "true"
    volumes:
      - ./onlyoffice/logs:/var/log/onlyoffice
      - ./onlyoffice/data:/var/www/onlyoffice/Data
      - ./onlyoffice/lib:/var/lib/onlyoffice
    networks: [onlyoffice_internal]
  callback-relay:
    image: ${CALLBACK_IMAGE:-docker.m.daocloud.io/nginx:alpine}
    restart: always
    command:
      - /bin/sh
      - -ec
      - |
        cat >/etc/nginx/conf.d/default.conf <<'EOF'
        server {
          listen 9000;
          location / {
            proxy_http_version 1.1;
            proxy_set_header Host $$host;
            proxy_set_header Connection "";
            proxy_pass http://unix:/run/fnoffice/app.sock:/;
          }
        }
        EOF
        exec nginx -g 'daemon off;'
    volumes:
      - ${FNOFFICE_APPDEST}:/run/fnoffice:ro
    networks: [onlyoffice_internal]
networks:
  onlyoffice_internal: {}
